    /// <summary>
    /// Caida de voltaje que la resistencia interna REAL le comeria a esta
    /// fuente con la corriente que esta entregando (I·R). Es SOLO informativa:
    /// el solver trata a la fuente como ideal (no se hunde bajo carga), pero
    /// este numero muestra cuanto caeria una fuente de verdad. La caida "en
    /// serio" — que afecte al resto del circuito — llega con el solver MNA.
    /// </summary>
    public double CaidaInternaEstimada =>
        IsBurnt ? 0 : Math.Abs(CorrienteEntregada) * ResistenciaInternaNominal;

    /// <summary>Voltaje que la fuente entregaria en sus bornes si su resistencia interna participara (V − I·R), estimado, nunca menor que 0.</summary>
    public double VoltajeConCargaEstimado =>
        Math.Max(0, Math.Abs(Voltage) - CaidaInternaEstimada);
